1package com.ibm.calcops.rules;
2
3import java.math.BigDecimal;
4import java.math.RoundingMode;
5
6/**
7 * Business Rule 2.3.2: Subtraction Policy with Negative Result Control.
8 */
9public final class SubtractionPolicyRule {
10
11 public static final String STATUS_OK = "00";
12 public static final String STATUS_OVERFLOW = "10";
13 public static final String STATUS_OUT_OF_RANGE = "40";
14 public static final String STATUS_NEGATIVE_RESULT = "50";
15
16 private static final BigDecimal MAX_OPERAND = new BigDecimal("5000000.00");
17 private static final BigDecimal MIN_OPERAND = new BigDecimal("-5000000.00");
18 private static final BigDecimal RESULT_LIMIT = new BigDecimal("1000000000");
19
20 public record Result(BigDecimal value, String statusCode) {}
21
22 // Direct Entry Point for Business Rule 2.3.2
23 public static Result evaluate(BigDecimal a, BigDecimal b, boolean allowNegative) {
24 // Step 1: Range validation
25 if (isOutOfRange(a) || isOutOfRange(b)) {
26 return new Result(BigDecimal.ZERO, STATUS_OUT_OF_RANGE);
27 }
28
29 // Step 2: Equal operand shortcut
30 if (a.compareTo(b) == 0) {
31 return new Result(BigDecimal.ZERO.setScale(2, RoundingMode.UNNECESSARY), STATUS_OK);
32 }
33
34 // Step 3: Difference & overflow check
35 BigDecimal diff = a.subtract(b).setScale(2, RoundingMode.DOWN);
36 if (diff.abs().compareTo(RESULT_LIMIT) >= 0) {
37 return new Result(BigDecimal.ZERO, STATUS_OVERFLOW);
38 }
39
40 // Step 4: Enforce negative result policy
41 if (diff.signum() < 0 && !allowNegative) {
42 return new Result(diff, STATUS_NEGATIVE_RESULT);
43 }
44
45 return new Result(diff, STATUS_OK);
46 }
47
48 private static boolean isOutOfRange(BigDecimal val) {
49 return val == null || val.compareTo(MAX_OPERAND) > 0 || val.compareTo(MIN_OPERAND) < 0;
50 }
51}
